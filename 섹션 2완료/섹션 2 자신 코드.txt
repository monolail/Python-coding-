<대표값>


N = int(input())

a = list(map(int, input().split()))

avg = 0

for i in range(N) :
    avg += a[i]

avg = int(avg/N + 0.5)

result = []

for i in range(N) :
    result.append(abs(a[i] - avg))

m = result[0]
location = 0
for i in range(N) :
    if result[i] == m :
        if a[i] > a[location] :
            location = i
            m = result[i]
    if result[i] < m :
        location = i
        m = result[i]

print(avg, location+1)


<정다면체>
import sys
sys.stdin=open("input.txt", "r")

N,M = map(int, input().split())

a = []

for i in range(1,N+1):
    for j in range(1,M+1) :
        a.append(i+j)

counts = {}

for item in a :
    if item in counts :
        counts[item] +=1

    else :
        counts[item] = 1


max_val = max(counts.values())

result = []

for key, value in counts.items() :
    if value == max_val :

        result.append(key)

for i in result :

    print(i, end = ' ')

import sys
sys.stdin=open("input.txt", "r")


N,M = map(int, input().split())

cnt = [0]*(N+M+3)

for i in range(1,N+1) :
    for j in range(1,M+1) :
        cnt[i+j] += 1

Large = cnt[0]

for i in range(len(cnt)) :
    if Large < cnt[i] :
        Large = cnt[i]

result = []

for i in range(len(cnt)) :
    if cnt[i] == Large :
        result.append(i)

for i in result :
    print(i, end = ' ')

import sys
sys.stdin=open("input.txt", "r")

def digit_sum(x) :

    k = str(x)
    sum = 0
    for i in k :
        sum += int(i)

    return sum

N = int(input())

number = list(map(int, input().split()))

result = []

for i in number :
    
    gap = digit_sum(i)

    result.append(gap)

max_val = max(result)

final = []

for i in range(N) :
    if result[i] == max_val :

        final.append(number[i])

<뒤집은 소수>
import sys
sys.stdin=open("input.txt", "r")

N = int(input())

a = list(map(str,input().split()))

def reverse(x) :

    rev = x[::-1]

    return int(rev)

def isPrime(x) :

    if x == 1:
        return False

    for i in range(2,x) :

        if x%i == 0:
            return False
        
    return True

result = []


for i in a :

    gap = reverse(i)

    result.append(gap)

for i in result :

    if isPrime(i) :
        print(i, end = ' ')

import sys
sys.stdin=open("input.txt", "r")

N = int(input())

result = []

for i in range(N) :
    a = list(map(int,input().split()))

    if a[0] == a[1] == a[2] :

        gap = 10000 + a[0] * 1000

        result.append(gap)

    elif a[0] == a[1]  :

        gap = 1000 + a[0] * 100

        result.append(gap)

    elif a[0] == a[2] :

        gap = 1000 + a[0] * 100

        result.append(gap)

    elif a[1] == a[2] :

        gap = 1000 + a[1] * 100

        result.append(gap)

    else :
        result.append(max(a) * 100)

print(max(result))
