def binary_search(array, target, start, end):
    while start <= end :
        mid = (start + end) // 2

        if array[mid] == target :
            return mid
    
        elif array[mid] > target :
            end = mid - 1
        
        else :
            start = mid + 1

    return None

def Count(capacity) :
    cnt = 1
    hap =0
    for x in L :
        if hap + x > capacity :
            cnt +=1
            hap = x
        else :
            hap += x
    return cnt

L = list(map(int,input().split()))
largest = max(L)
lt = 1
rt = sum(L)
res=0

while (lt <= rt) :
