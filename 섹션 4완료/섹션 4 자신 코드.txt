def binary_search(array, target, start, end):
    while start <= end :
        mid = (start + end) // 2

        if array[mid] == target :
            return mid
    
        elif array[mid] > target :
            end = mid - 1
        
        else :
            start = mid + 1

    return None

import sys
sys.stdin=open("input.txt", "r")

N = int(input())

meeting = []

for i in range(N) :
    s,e = map(int,input().split())
    meeting.append((s,e))

# 끝나는 시간대로 정렬
meeting.sort(key = lambda  x :(x[1],x[0]))

et = 0
cnt = 0

for s,e in meeting :
    if s >= et :
        cnt += 1
        et = e
print(cnt)

import sys
sys.stdin=open("input.txt", "r")

N = int(input())

Profile = []

for i in range(N) :
    H,W = map(int,input().split())
    Profile.append((H,W))

# 키는 이제 고려X
# 가장 키 큰 사람을 제외함으로써, 몸무게로만 승부
Profile.sort(reverse = True)

Heavy = 0
cnt = 0
for x,y in Profile :
    if y >= Heavy :
        cnt +=1
        Heavy = y

print(cnt)
