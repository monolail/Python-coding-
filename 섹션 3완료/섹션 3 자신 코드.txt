# ver2 : Ascii 활용
import sys
sys.stdin=open("input.txt", "r")

N = str(input())

# ascii code 변환 : 숫자 0~9 (48 ~ 57)
tmp = ''
for i in N :
    
    if ord(i) >= 48 and ord(i) <= 57 :
        tmp += i
    
gap = int(tmp)

cnt = 0

for i in range(1,gap+1) :
    if gap % i == 0 :
        cnt+=1

print(gap)
print(cnt)

N,M = map(int,input().split())

a = list(map(int,input().split()))

lt = 0
rt = 1
tot = a[0]
cnt = 0
while True :
    if tot < M :
        if rt < N :
            tot += a[rt]
            rt +=1
        else :
            break
    elif tot == M :
        cnt +=1
        tot-=a[lt]
        lt+=1
    else :
        tot -=a[lt]
        lt+=1

print(cnt)

# all() 함수는 리스트, 튜플 같은 반복 가능한 객체(iterable)의 모든 요소가 참(True)일 때만 True를 반환하고, 하나라도 거짓(False)
for i in range(1,N+1) :
